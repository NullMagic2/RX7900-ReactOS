function(rx7900_cpp_target target)
    set_target_properties(${target} PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED YES
        CXX_EXTENSIONS NO)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /WX /permissive-)
        target_compile_definitions(${target} PRIVATE _CRT_SECURE_NO_WARNINGS)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
    if(RX7900_SANITIZERS)
        if(MSVC OR CMAKE_CROSSCOMPILING OR COMMAND set_module_type)
            message(FATAL_ERROR "ASan/UBSan checks require a native GCC/Clang host build")
        endif()
        target_compile_options(${target} PRIVATE -fsanitize=address,undefined
            -fno-omit-frame-pointer)
        target_link_options(${target} PRIVATE -fsanitize=address,undefined)
    endif()
endfunction()

# Only pure CPU unit tests use this helper. Guest service tests stay opt-in.
function(rx7900_unit_test test_name target source library)
    add_executable(${target} ${source})
    target_link_libraries(${target} PRIVATE ${library})
    target_include_directories(${target} PRIVATE src/common)
    if(MINGW)
        target_link_options(${target} PRIVATE -static -static-libgcc -static-libstdc++)
    endif()
    rx7900_cpp_target(${target})
    add_test(NAME ${test_name} COMMAND ${target})
endfunction()
